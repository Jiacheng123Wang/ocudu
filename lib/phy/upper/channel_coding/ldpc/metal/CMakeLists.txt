# Build wiring for the Metal GPU LDPC decoder. The SynchroPlus sources in this
# directory are copied verbatim and must not be modified (see PLAN.md).
#
# Two shader-init paths are provided because the offline Metal shader compiler
# (metal/metallib) ships inside Xcode and is absent on Command-Line-Tools-only
# machines:
#   1. Offline-compiled metallib (preferred, built when the toolchain is found).
#   2. Embedded shader source for runtime compilation (newLibraryWithSource),
#      always generated - no external tool required.

# ---- Embedded shader source (runtime-compile fallback) ----
# Generated at configure time from the .metal sources; re-run cmake after editing them.
set(METAL_DECODER_SHADER ${CMAKE_CURRENT_SOURCE_DIR}/decode/ldpc_gpu_decoder.metal)
set(METAL_DECODER_SOURCE_HEADER ${CMAKE_CURRENT_BINARY_DIR}/ldpc_gpu_decoder_metal_source.h)
file(READ ${METAL_DECODER_SHADER} METAL_DECODER_SHADER_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1, " METAL_DECODER_SHADER_ARRAY "${METAL_DECODER_SHADER_HEX}")
file(WRITE ${METAL_DECODER_SOURCE_HEADER}
"// Auto-generated by CMake from decode/ldpc_gpu_decoder.metal: shader source for runtime
// compilation (newLibraryWithSource) when no offline-compiled metallib is available.
#pragma once

namespace ocudu {
static constexpr unsigned char ldpc_gpu_decoder_metal_source[] = { ${METAL_DECODER_SHADER_ARRAY} 0x00 };
}
")

# ---- Offline metallib (only when the Xcode Metal toolchain is present) ----
find_program(XCRUN_TOOL NAMES xcrun)
set(METAL_DECODER_METALLIB ${CMAKE_CURRENT_BINARY_DIR}/ocudu_ldpc_decoder.metallib)
set(METAL_TOOL_FOUND FALSE)
if(XCRUN_TOOL)
    execute_process(COMMAND ${XCRUN_TOOL} -sdk macosx -f metal
                    OUTPUT_VARIABLE METAL_TOOL_PATH OUTPUT_STRIP_TRAILING_WHITESPACE
                    RESULT_VARIABLE METAL_TOOL_RESULT ERROR_QUIET)
    if(METAL_TOOL_RESULT EQUAL 0)
        set(METAL_TOOL_FOUND TRUE)
        add_custom_command(
            OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/ocudu_ldpc_decoder.air
            COMMAND ${XCRUN_TOOL} -sdk macosx metal -c ${METAL_DECODER_SHADER}
                    -o ${CMAKE_CURRENT_BINARY_DIR}/ocudu_ldpc_decoder.air
            DEPENDS ${METAL_DECODER_SHADER}
            COMMENT "Compiling the Metal LDPC decoder shader (.metal -> .air)")
        add_custom_command(
            OUTPUT ${METAL_DECODER_METALLIB}
            COMMAND ${XCRUN_TOOL} -sdk macosx metallib ${CMAKE_CURRENT_BINARY_DIR}/ocudu_ldpc_decoder.air
                    -o ${METAL_DECODER_METALLIB}
            DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/ocudu_ldpc_decoder.air
            COMMENT "Packaging the Metal LDPC decoder shader (.air -> .metallib)")
        add_custom_target(ocudu_ldpc_decoder_metallib ALL DEPENDS ${METAL_DECODER_METALLIB})
        message(STATUS "Metal shader compiler found: offline metallib will be built")
    else()
        message(STATUS "Metal shader compiler not available (Xcode not installed): "
                       "runtime shader compilation will be used")
    endif()
endif()

# ---- Static library with the SynchroPlus GPU decoder implementation ----
enable_language(OBJCXX)
add_library(ocudu_ldpc_metal STATIC
    decode/ldpc_gpu_decoder.mm          # SynchroPlus copy (verbatim, manual reference counting)
    ocudu_metal_decoder_engine.mm       # ocudu-side engine driver (ARC)
    ldpc_decoder_metal.cpp)             # ocudu ldpc_decoder adapter
set_source_files_properties(decode/ldpc_gpu_decoder.mm PROPERTIES LANGUAGE OBJCXX)
set_source_files_properties(ocudu_metal_decoder_engine.mm PROPERTIES LANGUAGE OBJCXX COMPILE_OPTIONS "-fobjc-arc")
set_property(TARGET ocudu_ldpc_metal PROPERTY OBJCXX_STANDARD 17)
target_compile_options(ocudu_ldpc_metal PRIVATE -Wno-deprecated-declarations)
target_include_directories(ocudu_ldpc_metal PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}       # ldpc_decoder_metal.h / ocudu_metal_decoder_engine.h
    ${CMAKE_CURRENT_SOURCE_DIR}/decode
    ${CMAKE_CURRENT_SOURCE_DIR}/..   # ldpc_graph_impl.h (H-matrix generation from the protograph)
    ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(ocudu_ldpc_metal PUBLIC
    OCUDU_METAL_DECODER_METALLIB_PATH="${METAL_DECODER_METALLIB}")
# The adapter generates the packed H matrix from the ocudu ldpc_graph implementation.
target_link_libraries(ocudu_ldpc_metal PUBLIC ocudu_ldpc)
target_link_libraries(ocudu_ldpc_metal PRIVATE "-framework Metal" "-framework Foundation")

# ---- Unit-level validation binary (golden H check + decode parity vs the CPU decoder) ----

target_link_libraries(ldpc_metal_unit_test PRIVATE ocudu_channel_coding ocudu_ldpc_metal)

target_link_libraries(ldpc_metal_bler_test PRIVATE ocudu_channel_coding ocudu_ldpc_metal)
