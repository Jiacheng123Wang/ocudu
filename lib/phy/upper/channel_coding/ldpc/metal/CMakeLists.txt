# Build wiring for the Metal GPU LDPC decoder (layered normalized min-sum, see PLAN.md).
#
# The shader source is embedded at configure time and compiled at runtime with
# newLibraryWithSource - no Xcode/Metal toolchain is required at build time.
# Re-run cmake after editing the .metal file.

# ---- Embedded shader source (runtime compilation) ----
set(NMSL_DECODER_SHADER ${CMAKE_CURRENT_SOURCE_DIR}/ocudu_nms_layered_decoder.metal)
set(NMSL_DECODER_SOURCE_HEADER ${CMAKE_CURRENT_BINARY_DIR}/ocudu_nms_layered_decoder_metal_source.h)
file(READ ${NMSL_DECODER_SHADER} NMSL_DECODER_SHADER_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1, " NMSL_DECODER_SHADER_ARRAY "${NMSL_DECODER_SHADER_HEX}")
# Flooding NMS shader (ocudu-side): same treatment.
set(NMSF_DECODER_SHADER ${CMAKE_CURRENT_SOURCE_DIR}/ocudu_nms_flooding_decoder.metal)
set(NMSF_DECODER_SOURCE_HEADER ${CMAKE_CURRENT_BINARY_DIR}/ocudu_nms_flooding_decoder_metal_source.h)
file(READ ${NMSF_DECODER_SHADER} NMSF_DECODER_SHADER_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1, " NMSF_DECODER_SHADER_ARRAY "${NMSF_DECODER_SHADER_HEX}")
file(WRITE ${NMSF_DECODER_SOURCE_HEADER}
"// Auto-generated by CMake from ocudu_nms_flooding_decoder.metal: shader source for
// runtime compilation (newLibraryWithSource).
#pragma once

namespace ocudu {
static constexpr unsigned char ocudu_nms_flooding_decoder_metal_source[] = { ${NMSF_DECODER_SHADER_ARRAY} 0x00 };
}
")

file(WRITE ${NMSL_DECODER_SOURCE_HEADER}
"// Auto-generated by CMake from ocudu_nms_layered_decoder.metal: shader source for runtime
// compilation (newLibraryWithSource).
#pragma once

namespace ocudu {
static constexpr unsigned char ocudu_nms_layered_decoder_metal_source[] = { ${NMSL_DECODER_SHADER_ARRAY} 0x00 };
}
")

# ---- Static library: engine driver + ldpc_decoder adapter ----
enable_language(OBJCXX)
add_library(ocudu_ldpc_metal STATIC
    ocudu_metal_decoder_engine.mm       # engine driver (ARC)
    ldpc_decoder_metal.cpp)             # ocudu ldpc_decoder adapter
set_source_files_properties(ocudu_metal_decoder_engine.mm PROPERTIES LANGUAGE OBJCXX COMPILE_OPTIONS "-fobjc-arc")
set_property(TARGET ocudu_ldpc_metal PROPERTY OBJCXX_STANDARD 17)
target_compile_options(ocudu_ldpc_metal PRIVATE -Wno-deprecated-declarations)
target_include_directories(ocudu_ldpc_metal PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}       # ldpc_decoder_metal.h / ocudu_metal_decoder_engine.h
    ${CMAKE_CURRENT_SOURCE_DIR}/..   # ldpc_graph_impl.h (H-matrix generation from the protograph)
    ${CMAKE_CURRENT_BINARY_DIR})
# The adapter generates the packed H matrix from the ocudu ldpc_graph implementation.
target_link_libraries(ocudu_ldpc_metal PUBLIC ocudu_ldpc)
target_link_libraries(ocudu_ldpc_metal PRIVATE "-framework Metal" "-framework Foundation")

# ---- Unit-level validation binary (golden H check + decode parity vs the CPU decoder) ----
add_executable(ldpc_metal_unit_test test/ldpc_metal_unit_test.cpp)
target_link_libraries(ldpc_metal_unit_test PRIVATE ocudu_channel_coding ocudu_ldpc_metal)

# ---- Pure LDPC BLER benchmark (CPU vs GPU across SNRs and code rates) ----
add_executable(ldpc_metal_bler_test test/ldpc_metal_bler_test.cpp)
target_link_libraries(ldpc_metal_bler_test PRIVATE ocudu_channel_coding ocudu_ldpc_metal)
